73def build_runners(args) -> list: 

74 """Erzeugt die SolverRunner; nicht startbare Solver werden gemeldet und übersprungen.""" 

75 commands = [] 

76 if args.solver_path.strip(): 

77 commands.append(args.solver_path) 

78 commands.extend(args.with_solver) 

79 runners = [SolverRunner(cmd) for cmd in commands] 

80 if args.with_reference: 

81 runners.append(SolverRunner(REFERENCE_SOLVER_CMD, label="reference-cdcl")) 

82 labels = [r.label for r in runners] 

83 dupes = {l for l in labels if labels.count(l) > 1} 

84 if dupes: 

85 raise SystemExit(f"Solver-Labels nicht eindeutig: {sorted(dupes)}") 

86 return runners
